-- ETL：売上日時点で有効な顧客キーを引き当ててファクトに埋め込む
SELECT f.伝票番号, f.売上日, d.顧客キー, f.金額
FROM  stg.売上明細 AS f
LEFT JOIN dw.Dim_顧客 AS d
       ON  d.顧客ID  = f.顧客ID
       AND f.売上日 >= d.有効開始
       AND f.売上日 <  d.有効終了;

```m
// スナップショット履歴（スナップ日, 顧客ID, 顧客名, ランク）から Type2 を組む
let
    並べ替え = Table.Sort(スナップショット履歴,
        {{"顧客ID", Order.Ascending}, {"スナップ日", Order.Ascending}}),
    // 履歴を残す列だけをハッシュにまとめる。ここに入れた列の変化だけが行を増やす
    ハッシュ付き = Table.AddColumn(並べ替え, "属性ハッシュ",
        each Text.Combine({[顧客名], [ランク]}, "|"), type text),
    Idx = Table.AddIndexColumn(ハッシュ付き, "i", 0, 1, Int64.Type),
    // 前行と比較して変化点だけを残す
    変化点 = Table.SelectRows(Idx, (r) =>
        let 前 = Table.SelectRows(Idx, each [i] = r[i] - 1)
        in  Table.IsEmpty(前) or 前{0}[顧客ID] <> r[顧客ID]
            or 前{0}[属性ハッシュ] <> r[属性ハッシュ]),
    再採番 = Table.AddIndexColumn(変化点, "行番号", 0, 1, Int64.Type),
    // 有効終了 = 次の変化点の前日。無ければ番兵値
    有効終了 = Table.AddColumn(再採番, "有効終了", (r) =>
        let 次 = Table.SelectRows(再採番,
                each [顧客ID] = r[顧客ID] and [行番号] = r[行番号] + 1)
        in  if Table.IsEmpty(次) then #date(9999,12,31)
            else Date.AddDays(次{0}[スナップ日], -1), type date),
    改名 = Table.RenameColumns(有効終了, {{"スナップ日", "有効開始"}}),
    現在 = Table.AddColumn(改名, "現在フラグ",
        each if [有効終了] = #date(9999,12,31) then "Yes" else "No", type text),
    SK = Table.AddIndexColumn(現在, "顧客キー", 1, 1, Int64.Type)
in
    Table.SelectColumns(SK,
        {"顧客キー","顧客ID","顧客名","ランク","有効開始","有効終了","現在フラグ"})
```

> [!TRAP] つまずきポイント
> 各行から `Table.SelectRows` を呼ぶため計算量は行数の2乗。数十万行では実用外。
> 大規模ならソース側の `LAG` / `LEAD` で処理する。

## 実装後の検証と設計判断

| 検証項目 | 方法 | 期待結果 |
|---|---|---|
| 期間の重複 | 同じ顧客IDで有効期間が重なる行 | 0件 |
| 期間の欠落 | 前行の有効終了 + 1日 = 次行の有効開始 | すべて連続 |
| 現在行の一意性 | 顧客IDごとの現在フラグ = Yes | すべて1件 |
| サロゲートキーの一意性 | 顧客キーの重複 | 0件 |
| ファクトの引き当て漏れ | 顧客キーが null または不明キー | 0件、または既知の件数 |
| 総計の一致 | 導入前後の全期間売上合計 | 完全に一致。ずれるなら引き当て漏れか二重計上 |

- [ ] 属性ごとに履歴を残すか上書きするかを決めたか
- [ ] 「当時の姿」と「今の姿」のどちらが既定かを利用部門と合意したか
- [ ] ディメンションのキーはサロゲートキーか
- [ ] ファクトはサロゲートキーを持っているか
- [ ] 引き当ては取り込み時に1度だけ行っているか
- [ ] 有効期間に重複・欠落がないか検証したか
- [ ] 現在フラグが顧客IDごとに1件だけか検証したか
- [ ] 導入前後で総計が一致することを確認したか
- [ ] 列名に「当時」「現在」を明記したか

## 理解度チェック

1. 営業所の管轄が再編された。「新管轄で全期間」と「当時の管轄」の両立をどう設計するか。
