購入率 := DIVIDE( DISTINCTCOUNT( Fact_イベント[購入ユーザーID] ), DISTINCTCOUNT( Dim_割当[ユーザーID] ) )
購入率_対照群 := CALCULATE( [購入率], Dim_割当[群] = "A" )
購入率_処置群 := CALCULATE( [購入率], Dim_割当[群] = "B" )
絶対リフト := [購入率_処置群] - [購入率_対照群]

差の標準誤差 :=
VAR pA = [購入率_対照群]
VAR pB = [購入率_処置群]
VAR nA = CALCULATE( DISTINCTCOUNT( Dim_割当[ユーザーID] ), Dim_割当[群] = "A" )
VAR nB = CALCULATE( DISTINCTCOUNT( Dim_割当[ユーザーID] ), Dim_割当[群] = "B" )
RETURN SQRT( DIVIDE( pA * ( 1 - pA ), nA ) + DIVIDE( pB * ( 1 - pB ), nB ) )

差_下限95 := [絶対リフト] - 1.96 * [差の標準誤差]
差_上限95 := [絶対リフト] + 1.96 * [差の標準誤差]
