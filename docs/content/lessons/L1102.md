(税抜金額 as number, optional 税率 as number) as number =>
let
    率 = if 税率 = null then 0.10 else 税率,

in
    丸め

```m
// fnファイル変換：Binary を受け取ってテーブルを返す
(中身 as binary) as table =>
let
    Csv = Csv.Document(中身, [Delimiter=",", Encoding=65001, QuoteStyle=QuoteStyle.Csv]),
    Head = Table.PromoteHeaders(Csv, [PromoteAllScalars=true]),
    Typed = Table.TransformColumnTypes(Head, {{"売上日", type date}, {"金額", Currency.Type}})
in
    Table.SelectRows(Typed, each [売上日] <> null)
```

```m
// Fact_売上：フォルダの全CSVに関数を適用する
let
    Src = Table.SelectRows(Folder.Files(pDataPath), each Text.Lower([Extension]) = ".csv"),
    除外 = Table.SelectRows(Src, each not Text.StartsWith([Name], "~$")),   // 一時ファイル
    適用 = Table.AddColumn(除外, "データ", each fnファイル変換([Content])),
    取込元付き = Table.SelectColumns(適用, {"Name", "データ"}),              // 由来列を残す
    展開 = Table.ExpandTableColumn(取込元付き, "データ", {"売上日", "金額"})
in
    展開
```

> [!TRAP] Folder.Files はサブフォルダも拾う
> アーカイブ配下まで毎回読む。階層を絞り、`~$` の一時ファイルを除外する。由来列は必ず残す。

## 命名規約とまとめ

| 接頭辞 | 対象 | 例 |
|---|---|---|
| `fn` | カスタム関数 | `fnファイル変換` |
| `p` | パラメータ | `pServer`, `RangeStart` |
| `stg_` | ステージング（読み込み無効） | `stg_売上生データ` |
| `Dim_` / `Fact_` | モデルに読み込む | `Fact_売上` |
| `chk_` | 品質チェック（読み込み無効） | `chk_マスタ未登録商品` |

- グループは「パラメータ / 関数 / ステージング / モデル / チェック」の5つで始める
- 関数を直すと呼び出し元すべてが同時に変わる。依存関係ビューで影響先を確認する
- `DateTime.LocalNow()` はフィルター条件に使わない。冪等性が壊れる

> [!EXAM] 複数CSVの統合はフォルダ接続
> `Folder.Files` ＋ファイルの変換関数が正解。1本ずつ接続して追加、は誤答の定番。
